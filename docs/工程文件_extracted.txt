二、 硬件与环境级“排雷指南”（前置必读）
在 Windows 本地混跑这 7 个阶段，以下三个技术雷区最容易卡死进度：
CUDA 与新一代显卡架构（Blackwell sm_120）的编译雷区
现状：如果你使用的是最新一代显卡（如 RTX 50 系列），必须安装 CUDA 12.8+ 及配套的 PyTorch Nightly / 2.7+ (cu128)。
Phase 3 (TRELLIS) 强烈建议直接上 WSL2 (Ubuntu 22.04/24.04)：不要在原生 Windows 上硬磕 TRELLIS 的 spconv、kaolin、nvdiffrast 和 diffoctreerast 编译，Windows 下的 MSVC 编译器版本与 CUDA 扩展极其容易冲突。将 TRELLIS 封装在 WSL2 内部，写一个 20 行的 FastAPI 接口暴露在 localhost:8003，主控脚本在 Windows 原生环境通过 HTTP 传图拿回 .glb，彻底解耦且性能零损耗。
96GB 级大内存条件下的 VRAM（显存）红绿灯调度
单张高端显卡通常为 16GB~24GB 显存。如果 Ollama（常驻 8~10GB）、ComfyUI（FLUX 峰值 12~16GB）和 TRELLIS（峰值 14GB）同时霸占显存，必然触发 OOM 或降级到系统共享内存导致速度暴跌 10 倍。
解决策略：利用充裕的系统内存做缓冲，在总控脚本中执行严格的“显存卸载指令”——
调完 Ollama 后，发送 keep_alive: 0 立即释放显存；
调完 ComfyUI 后，请求其 /free 接口 ({"unload_models": true, "free_memory": true}) 清空显存，再唤醒 TRELLIS。
Phase 5 (FreeCAD + CadQuery) 的自愈循环（Self-Healing Loop）
大模型直接写 3D 参数化代码（无论是 freecad-mcp 还是 CadQuery），一次性通过率不到 40%（常见报错：倒角半径大于边长导致 BRep 拓扑崩溃、草图未闭合）。
核心机制：千万不要只让模型生成一次。必须在 Phase 5 写一个 while retry < 5: 循环，用 FreeCAD 的无头模式（freecadcmd.exe）执行生成的 Python 脚本，捕获 stderr 报错堆栈 + 导出模型的包围盒/体积数值，原封不动喂回给主控模型让它自己修代码，直到无报错产出合法 .step 为止。

import osimport ollamafrom qdrant_client import QdrantClientfrom qdrant_client.models import Distance, VectorParams, PointStructfrom FlagEmbedding import BGEM3FlagModel
# ================= 配置区 =================
QDRANT_URL = "http://localhost:6333"
COLLECTION_NAME = "mfg_knowledge_base"
EMBED_MODEL_PATH = "BAAI/bge-m3"  # 会自动缓存至 D:\hyperdesign\models\hf_cache
LLM_MODEL = "qwen3.5:9b"          # 或你拉取的本地模型名称
class HyperDesignBrain:
def __init__(self):
print("[1/3] 连接 Qdrant 向量库...")
self.qdrant = QdrantClient(url=QDRANT_URL)
self._init_collection()
print("[2/3] 加载 bge-m3 向量模型...")
self.embedder = BGEM3FlagModel(EMBED_MODEL_PATH, use_fp16=True)
def _init_collection(self):
collections = [c.name for c in self.qdrant.get_collections().collections]
if COLLECTION_NAME not in collections:
self.qdrant.create_collection(
collection_name=COLLECTION_NAME,
vectors_config=VectorParams(size=1024, distance=Distance.COSINE),
)
def ingest_documents(self, docs: list[dict]):
"""灌入行业工艺标准与历史设计规范"""
texts = [d["content"] for d in docs]
embeddings = self.embedder.encode(texts, batch_size=12, max_length=1024)["dense_vecs"]
points = [
PointStruct(id=idx, vector=emb.tolist(), payload=doc)
for idx, (emb, doc) in enumerate(zip(embeddings, docs))
]
self.qdrant.upsert(collection_name=COLLECTION_NAME, points=points)
print(f"成功入库 {len(points)} 条制造业知识条目。")
def query_and_plan(self, user_brief: str, unload_vram_after: bool = True) -> str:
"""检索私有工艺库并生成结构化设计约束"""
query_vec = self.embedder.encode([user_brief])["dense_vecs"][0].tolist()
hits = self.qdrant.search(
collection_name=COLLECTION_NAME,
query_vector=query_vec,
limit=3
)
context_str = "\n".join([f"- [{h.payload['source']}]: {h.payload['content']}" for h in hits])
system_prompt = (
"你是 HyperDesign 工业设计主控引擎。请严格依据【企业内部工艺规范】回答问题或拆解设计任务。\n"
"如果规范中有明确数值（如壁厚、拔模角、公差、材料牌号），必须原样引用，严禁捏造。"
)
prompt = f"【企业内部工艺规范】\n{context_str}\n\n【输入需求】\n{user_brief}"
# keep_alive=0 确保在流水线进入下一阶段（如FLUX/TRELLIS）前立即释放显存
keep_alive = 0 if unload_vram_after else "5m"
response = ollama.chat(
model=LLM_MODEL,
messages=[
{"role": "system", "content": system_prompt},
{"role": "user", "content": prompt}
],
keep_alive=keep_alive
)
return response["message"]["content"]

import osimport jsonimport timeimport uuidimport urllib.requestimport requestsimport ollamafrom pathlib import Path
# ================= 配置区 =================

LLM_MODEL = "qwen3.5:9b"
class Phase2ConceptExplorer:
def __init__(self, project_dir: Path):
self.project_dir = project_dir
self.concepts_dir = project_dir / "02_concepts"
self.concepts_dir.mkdir(parents=True, exist_ok=True)
self.manifest_path = project_dir / "manifest.json"
def brief_to_flux_prompts(self, design_brief: str, num_variants: int = 4) -> list[dict]:
"""Step 1: 调用本地主控模型，将中文工程 Brief 转化为多组差异化的 FLUX 英文提示词"""
system_prompt = """你是 HyperDesign 的工业视觉导演。你的任务是将用户的【产品设计需求与工艺约束】转化为用于 FLUX.1-dev 的英文提示词。
为了确保后续 TRELLIS 能够完美重建 3D 网格，每条提示词必须遵循以下铁律：
1. 视角固定：Single whole product, isometric three-quarter view, centered in frame, orthographic-like lens (no wide-angle distortion).
2. 背景与光照：Solid flat neutral grey background, studio softbox diffuse lighting, no harsh shadows, no reflections, matte surface finish.
3. 工程细节：Clearly visible parting lines, functional chamfers, screw bosses or snap-fit seams if applicable, realistic industrial manufacturing proportions.
4. 请输出严格的 JSON 数组格式，包含 4 个不同造型演化方向的变体（例如：硬朗切面风、圆润流线风、模块化装甲风、极简包豪斯风），字段为 [{"variant_name": "...", "flux_prompt": "..."}]。"""
response = ollama.chat(
model=LLM_MODEL,
format="json",
messages=[
{"role": "system", "content": system_prompt},
{"role": "user", "content": design_brief}
],
keep_alive=0  # 关键：生成完提示词立即卸载 Ollama 显存，把全部 VRAM 让给 FLUX
)
raw_json = json.loads(response["message"]["content"])
# 兼容模型返回 {"variants": [...]} 或直接返回 [...]
variants = raw_json if isinstance(raw_json, list) else raw_json.get("variants", list(raw_json.values())[0])
print(f"[Brain] 已生成 {len(variants)} 组结构化 FLUX 提示词，并已释放 Ollama 显存。")
return variants[:num_variants]
def run_comfy_batch(self, variants: list[dict], base_seed: int = 10000):
"""Step 2: 驱动 ComfyUI API 批量生成概念图并落盘到项目目录"""
with open(WORKFLOW_JSON_PATH, "r", encoding="utf-8") as f:
workflow_template = json.load(f)
generated_records = []
for idx, item in enumerate(variants):
workflow = json.loads(json.dumps(workflow_template))
seed = base_seed + idx * 101
# 注入正向提示词与随机种子（根据你导出的 API JSON 节点 ID 调整，通常 CLIPTextEncode 为 "6"，KSampler/RandomNoise 为 "25" 或 "31"）
for node_id, node in workflow.items():
if node.get("class_type") == "CLIPTextEncode":
node["inputs"]["text"] = item["flux_prompt"]
elif node.get("class_type") in ["RandomNoise", "KSampler"]:
if "noise_seed" in node["inputs"]:
node["inputs"]["noise_seed"] = seed
elif "seed" in node["inputs"]:
node["inputs"]["seed"] = seed
# 提交任务到 ComfyUI
prompt_id = self._queue_prompt(workflow)["prompt_id"]
print(f"[ComfyUI] 正在渲染方案 {idx+1}/{len(variants)} ({item.get('variant_name', 'Variant')}) | Seed: {seed}...")
# 轮询等待生成完成并下载图像
img_bytes = self._wait_and_fetch_image(prompt_id)
img_filename = f"concept_{idx+1:02d}_seed{seed}.png"
img_save_path = self.concepts_dir / img_filename
with open(img_save_path, "wb") as img_f:
img_f.write(img_bytes)
generated_records.append({
"id": f"concept_{idx+1:02d}",
"variant_name": item.get("variant_name", f"方案{idx+1}"),
"seed": seed,
"prompt": item["flux_prompt"],
"file": f"02_concepts/{img_filename}",
"selected_for_3d": False  # 留待设计师勾选，同时作为 Phase 7 的 DPO 数据源
})
# Step 3: 渲染完毕后强制清空 ComfyUI 显存，为 Phase 3 (TRELLIS) 腾出全部空间
self.free_comfy_vram()
self._update_manifest(generated_records)
print(f"[Phase 2 完成] 共生成 {len(generated_records)} 张概念图，保存在 {self.concepts_dir}")
def _queue_prompt(self, workflow: dict) -> dict:
data = json.dumps({"prompt": workflow, "client_id": str(uuid.uuid4())}).encode("utf-8")
req = urllib.request.Request(f"http://{COMFY_HOST}/prompt", data=data)
return json.loads(urllib.request.urlopen(req).read())
def _wait_and_fetch_image(self, prompt_id: str) -> bytes:
while True:
res = urllib.request.urlopen(f"http://{COMFY_HOST}/history/{prompt_id}")
history = json.loads(res.read())
if prompt_id in history:
outputs = history[prompt_id]["outputs"]
for node_output in outputs.values():
if "images" in node_output:
img_info = node_output["images"][0]
url = f"http://{COMFY_HOST}/view?filename={img_info['filename']}&subfolder={img_info['subfolder']}&type={img_info['type']}"
return urllib.request.urlopen(url).read()
time.sleep(1.0)

一、 架构避坑：为什么用“WSL2 跑 TRELLIS + Windows 原生跑 Blender”？

app = FastAPI(title="HyperDesign TRELLIS Service")
MODEL_DIR = "/mnt/d/hyperdesign/models/trellis/TRELLIS-image-large"
class ReconstructRequest(BaseModel):

def win_to_wsl_path(win_path: str) -> Path:
p = win_path.replace("\\", "/")
if p[1:3] == ":/":
drive = p[0].lower()
return Path(f"/mnt/{drive}/{p[3:]}")
return Path(p)
@app.post("/image_to_3d")def reconstruct_3d(req: ReconstructRequest):

out_path = win_to_wsl_path(req.win_output_glb)

五、 blender-mcp 在这一步怎么发挥最大价值？
很多人的误区是“让大模型通过 blender-mcp 从零捏一个产品外壳”，这在实际工程中效果极差。在 HyperDesign 架构里，分工非常清晰：
重活累活交给自动化流水线：TRELLIS 负责从 2D 概念图恢复复杂曲面与整体形态，blender_clean_mesh.py 负责校准毫米尺寸、坐平 Z=0 并统一工业灰材质。

二、 关键桥梁：从 Phase 3 白模自动提取“几何截面锚点”
如果只告诉大模型“生成一个 120×80×45mm 的传感器外壳”，它写出来的永远是一个呆板的火柴盒，完全丢失了 Phase 3 白模的工业造型比例（比如底宽上窄的阶梯感、顶部的收腰比例）。
因此，在让大模型写 CadQuery 代码前，我们先用 trimesh 对 Phase 3 的 white_model_clean.stl 在 Z 轴 10%（底座）、50%（腰线）、90%（顶盖） 三个高度做自动激光切片，量出精确的截面长宽与倒角包络，喂给大模型作为硬性几何约束：
Python
# D:\hyperdesign\src\utils_mesh_slicer.pyimport trimeshimport numpy as np
def extract_white_model_anchors(stl_path: str) -> dict:
"""对 Phase 3 清洗后的白模做 Z 轴三层高度切片，提取真实截面长宽作为 CAD 参数锚点"""
mesh = trimesh.load_mesh(stl_path)
min_bound, max_bound = mesh.bounds
total_dims = max_bound - min_bound  # [X, Y, Z]
anchors = {
"overall_bbox_mm": {
"x": round(float(total_dims[0]), 2),
"y": round(float(total_dims[1]), 2),
"z": round(float(total_dims[2]), 2)
},
"z_sections": {}
}
for label, ratio in [("bottom_10pct", 0.1), ("mid_50pct", 0.5), ("top_90pct", 0.9)]:
z_height = min_bound[2] + total_dims[2] * ratio
slice_2d = mesh.section(plane_origin=[0, 0, z_height], plane_normal=[0, 0, 1])
if slice_2d is not None:
slice_pts = slice_2d.vertices
sec_span = slice_pts.max(axis=0) - slice_pts.min(axis=0)
anchors["z_sections"][label] = {
"z_mm": round(float(z_height), 2),
"span_x_mm": round(float(sec_span[0]), 2),
"span_y_mm": round(float(sec_span[1]), 2)
}
return anchors
三、 核心引擎：带“自验证与自愈闭环”的 Text-to-CAD 生成器
下面这份核心脚本（D:\hyperdesign\src\phase5_cad_engine.py）实现了你要求的带自验证的参数化模型生成，并且内置了极为关键的 “报错/超差自动重试 + DPO 负样本自动捕获” 机制：
硬核几何校验（不靠肉眼，靠数学）：每次模型写出 Python 代码后，子进程在沙箱中运行它并检查 4 个硬指标：
脚本是否零报错退出？
是否成功产出了非空的 .step 文件？
OpenCascade 底层拓扑检验 solid.isValid() 是否为 True（无自交面、无破洞）？
生成实体的三维包围盒（Bounding Box）与你在 Phase 4 锁定的 target_bbox_mm 误差是否在 ±1.5mm 以内？
自愈（Self-Healing）与 DPO 数据沉淀：如果第 1 次生成因为 Fillet radius too large（倒角过大）报错，脚本会把报错堆栈和几何误差塞回给大模型，让它在第 2 轮修正代码。一旦第 2 轮通过，第 1 轮失败的代码作为 rejected、第 2 轮成功的代码作为 chosen，自动写入 Phase 7 的 CAD DPO 训练集！
Python
# D:\hyperdesign\src\phase5_cad_engine.pyimport jsonimport subprocessimport tempfileimport ollamafrom pathlib import Pathfrom utils_mesh_slicer import extract_white_model_anchors
LLM_MODEL = "qwen3.5:9b"
DATASET_DIR = Path(r"D:\hyperdesign\models\checkpoints\dpo_dataset")
DATASET_DIR.mkdir(parents=True, exist_ok=True)
SYSTEM_PROMPT = """你是 HyperDesign 首席参数化 CAD 工程师，精通 Python CadQuery (OpenCascade 内核)。
你的任务是根据【目标包络尺寸】、【白模多层截面比例】、【制造工艺约束】和【设计师批注】，编写可直接运行的 CadQuery 脚本。
必须遵守的工程与编码铁律：
1. 脚本顶部必须定义清晰的全局变量参数表（如 LENGTH_X, WIDTH_Y, HEIGHT_Z, WALL_THICKNESS, DRAFT_ANGLE, FILLET_R 等），方便人工二次修改。
2. 建模坐标系：XY 平面居中，底部基准面位于 Z=0（即首个基础拉伸体使用 .translate((0, 0, HEIGHT_Z/2)) 或在 Z>=0 空间构建）。
3. 严禁使用危险的大半径 fillet() 导致 BRep_API: command not done 拓扑崩溃；建议外圆角 R <= 3.0mm，内圆角 R <= 1.0mm，且在 shell() 抽壳之前先做外部主倒角。
4. 脚本末尾必须将最终主实体赋值给变量 `result_solid`（类型为 cq.Workplane），并使用 `cq.exporters.export(result_solid, STEP_OUTPUT_PATH)` 导出 STEP 文件。
5. 只输出纯 Python 代码（包裹在 ```python ... ``` 中），不要解释。"""
class Phase5ParametricCAD:
def __init__(self, project_dir: Path):
self.project_dir = project_dir
self.manifest_path = project_dir / "manifest.json"
self.cad_dir = project_dir / "05_cad"
self.cad_dir.mkdir(parents=True, exist_ok=True)
def generate_and_verify(self, max_retries: int = 5):
with open(self.manifest_path, "r", encoding="utf-8") as f:
manifest = json.load(f)
# 1. 提取白模多层截面尺寸锚点
stl_path = self.project_dir / manifest["mesh_metrics"]["clean_stl"]
mesh_anchors = extract_white_model_anchors(str(stl_path))
target_bbox = manifest["target_bbox_mm"]
mfg = manifest.get("manufacturing_constraints", {})
notes = manifest.get("human_feedback", {}).get("notes", "常规工业壳体，四角带M4安装柱，抽壳壁厚均匀")
step_out_path = (self.cad_dir / "parametric_model.step").as_posix()
base_task_prompt = f"""请编写 CadQuery 参数化建模脚本，并将 STEP 导出至路径："{step_out_path}"
【1. 强制目标物理包络 (mm)】(生成的模型外包围盒长宽高误差不得超过 ±1.5mm)：
- 总长 (X): {target_bbox['x']} mm
- 总宽 (Y): {target_bbox['y']} mm
- 总高 (Z): {target_bbox['z']} mm
【2. Phase 3 白模实测截面比例参考】：{json.dumps(mesh_anchors['z_sections'], ensure_ascii=False, indent=2)}
【3. 制造工艺与 DFM 约束】：
- 加工工艺: {mfg.get('process', '注塑/CNC')}
- 标准壁厚: {mfg.get('min_wall_thickness_mm', 2.4)} mm
- 设计师结构指令: {notes}
"""
messages = [
{"role": "system", "content": SYSTEM_PROMPT},
{"role": "user", "content": base_task_prompt}
]
failed_attempts = []
for attempt in range(1, max_retries + 1):
print(f"\n[Phase 5] 第 {attempt}/{max_retries} 轮生成参数化 CadQuery 脚本...")
resp = ollama.chat(model=LLM_MODEL, messages=messages, keep_alive="2m")
code_str = self._extract_python_code(resp["message"]["content"])
# 保存当前轮次脚本
script_path = self.cad_dir / f"cad_gen_v{attempt}.py"
script_path.write_text(code_str, encoding="utf-8")
# 执行 OpenCascade 硬件级几何校验
is_valid, verify_report = self._run_occ_verification(script_path, target_bbox)
if is_valid:
print(f"[Phase 5 验证通过!] 实体体积: {verify_report['volume_mm3']} mm³ | 实测包络: {verify_report['bbox_mm']}")
# 释放 Ollama 显存
ollama.chat(model=LLM_MODEL, messages=[], keep_alive=0)
# 将最终通过的代码固化为正式脚本
final_script = self.cad_dir / "parametric_model.py"
final_script.write_text(code_str, encoding="utf-8")
# 如果经历了报错修复才成功，自动把 (失败代码 vs 成功代码) 存入 Phase 7 DPO 数据集！
if failed_attempts:
self._record_cad_dpo(base_task_prompt, chosen_code=code_str, rejected_list=failed_attempts)
self._update_manifest(verify_report, final_script)
return True
else:
print(f"[Phase 5 校验未通过] 原因: {verify_report['error_msg']}")
failed_attempts.append(code_str)
# 将报错堆栈与包络偏差反馈给大模型进行自愈修正
messages.append({"role": "assistant", "content": f"```python\n{code_str}\n```"})
messages.append({
"role": "user",
"content": (
f"上一段代码执行或几何验证失败！具体诊断信息如下：\n{verify_report['error_msg']}\n"
"请修复上述拓扑报错或尺寸偏差（若因 fillet/shell 失败，请减小倒角半径或调整特征顺序），重新输出完整 Python 代码。"
)
})
raise RuntimeError(f"经过 {max_retries} 轮自愈循环仍未通过几何校验，请检查 prompt 约束或通过 freecad-mcp 介入。")
def _run_occ_verification(self, script_path: Path, target_bbox: dict) -> tuple[bool, dict]:
"""在沙箱后处理中挂载 OpenCascade 几何检查器：校验无报错、BRep 拓扑合法性、体积>0 以及三维尺寸精度"""
verifier_suffix = f"""
# === HyperDesign Automated OCC Geometry Verifier ===
import json as _json
_solid = result_solid.val()
_bbox = _solid.BoundingBox()
_report = {{
"is_valid_brep": bool(_solid.isValid()),
"volume_mm3": round(float(_solid.Volume()), 2),
"bbox_mm": {{
"x": round(float(_bbox.xlen), 2),
"y": round(float(_bbox.ylen), 2),
"z": round(float(_bbox.zlen), 2)
}}
}}
print("__VERIFY_JSON__" + _json.dumps(_report))
"""
full_test_code = script_path.read_text(encoding="utf-8") + "\n" + verifier_suffix
with tempfile.NamedTemporaryFile("w", suffix=".py", delete=False, encoding="utf-8") as tmp:
tmp.write(full_test_code)
tmp_path = tmp.name
proc = subprocess.run(["python", tmp_path], capture_output=True, text=True, timeout=60)
Path(tmp_path).unlink(missing_ok=True)
if proc.returncode != 0:
return False, {"error_msg": f"Python 运行时/OCC 内核报错:\n{proc.stderr[-800:]}"}
# 解析校验输出
for line in proc.stdout.splitlines():
if line.startswith("__VERIFY_JSON__"):
report = json.loads(line.replace("__VERIFY_JSON__", ""))
if not report["is_valid_brep"] or report["volume_mm3"] <= 1.0:
return False, {"error_msg": f"BRep 实体拓扑非法或体积为零: {report}"}
# 校验长宽高偏差是否在 ±1.5mm 以内
bx, by, bz = report["bbox_mm"]["x"], report["bbox_mm"]["y"], report["bbox_mm"]["z"]
tx, ty, tz = target_bbox["x"], target_bbox["y"], target_bbox["z"]
if abs(bx - tx) > 1.5 or abs(by - ty) > 1.5 or abs(bz - tz) > 1.5:
return False, {
"error_msg": f"生成的 CAD 实体外尺寸 ({bx}x{by}x{bz} mm) 与目标包络 ({tx}x{ty}x{tz} mm) 偏差超过 1.5mm！"
}
return True, report
return False, {"error_msg": "脚本未能输出 __VERIFY_JSON__，请确认最终实体变量名严格为 `result_solid`。"}
@staticmethod
def _extract_python_code(text: str) -> str:
if "```python" in text:
return text.split("```python")[1].split("```")[0].strip()
elif "```" in text:
return text.split("```")[1].split("```")[0].strip()
return text.strip()
def _record_cad_dpo(self, prompt: str, chosen_code: str, rejected_list: list[str]):
dpo_file = DATASET_DIR / "phase5_cad_code_dpo.jsonl"
with open(dpo_file, "a", encoding="utf-8") as f:
for rej in rejected_list:
f.write(json.dumps({
"prompt": prompt,
"chosen": chosen_code,
"rejected": rej
}, ensure_ascii=False) + "\n")
print(f"[DPO 收集器] 已自动捕获 {len(rejected_list)} 组 CAD 代码正负偏好样本对！")
def _update_manifest(self, report: dict, script_path: Path):
with open(self.manifest_path, "r", encoding="utf-8") as f:
manifest = json.load(f)
manifest["cad_verification"] = {
"script_file": f"05_cad/{script_path.name}",
"step_file": "05_cad/parametric_model.step",
"is_valid_brep": report["is_valid_brep"],
"volume_mm3": report["volume_mm3"],
"verified_bbox_mm": report["bbox_mm"]
}
with open(self.manifest_path, "w", encoding="utf-8") as f:
json.dump(manifest, f, ensure_ascii=False, indent=2)

四、 产出样板：大模型生成的工业级参数化代码长什么样？
通过上述 Phase5ParametricCAD 引擎跑通后，落在你 05_cad/parametric_model.py 里的不再是一次性的死模型，而是结构工程师打开就能看懂、改顶部几个数字就能瞬間派生出新规格系列件的标准参数化资产：
Python
import cadquery as cq
# ================= 1. 工程参数驱动表 (改数值即可全局联动) =================
LENGTH_X = 120.0          # 外壳总长 (mm)
WIDTH_Y = 80.0            # 外壳总宽 (mm)
HEIGHT_Z = 45.0           # 外壳总高 (mm)
WALL_THICKNESS = 2.4      # 二号车间 PC+ABS 规范壁厚 (mm)
OUTER_FILLET_R = 4.0      # 四周主棱边圆角 (mm)
TOP_CHAMFER = 1.5         # 顶部抗冲击C角 (mm)
BOSS_OD = 8.0             # 四角 M4 螺钉柱外径 (mm)
BOSS_ID = 3.4             # M4 自攻底孔内径 (mm)
BOSS_INSET = 8.0          # 螺钉柱距外壁中心距 (mm)
WINDOW_DIA = 8.0          # 顶部状态指示灯导光孔径 (mm)
STEP_OUTPUT_PATH = "D:/hyperdesign/workspace/proj_20261014_001/05_cad/parametric_model.step"
# ================= 2. 特征树构建流程 =================# Step A: 基础包络体构建（底部坐平 Z=0）
base_body = (
cq.Workplane("XY")
.box(LENGTH_X, WIDTH_Y, HEIGHT_Z, centered=(True, True, False))
.edges("|Z")
.fillet(OUTER_FILLET_R)
.edges(">Z")
.chamfer(TOP_CHAMFER)
)
# Step B: 底部抽壳形成均匀 2.4mm 壁厚腔体
hollow_shell = base_body.faces("<Z").shell(-WALL_THICKNESS)
# Step C: 内部四角生成 M4 安装螺钉柱（从内顶面延伸至底面）
boss_x = LENGTH_X / 2.0 - BOSS_INSET
boss_y = WIDTH_Y / 2.0 - BOSS_INSET
boss_pts = [(-boss_x, -boss_y), (boss_x, -boss_y), (boss_x, boss_y), (-boss_x, boss_y)]
shell_with_bosses = (
hollow_shell.faces(">Z").workplane(offset=-WALL_THICKNESS)
.pushPoints(boss_pts)
.circle(BOSS_OD / 2.0)
.extrude(-(HEIGHT_Z - WALL_THICKNESS), combine=True)
.faces("<Z").workplane()
.pushPoints(boss_pts)
.hole(BOSS_ID, depth=HEIGHT_Z - WALL_THICKNESS - 2.0)
)
# Step D: 顶面加工导光柱安装通孔
result_solid = (
shell_with_bosses.faces(">Z").workplane()
.center(LENGTH_X * 0.25, 0)
.hole(WINDOW_DIA)
)
# 导出标准 STEP AP214 工业交换格式
cq.exporters.export(result_solid, STEP_OUTPUT_PATH)

二、 核心设计：把“国标机械制图规范（GB/T）”写进代码
为什么很多 AI 转出来的 DXF 车间师傅看一眼就骂街？因为所有线条全糊在一个默认的 0 图层上，粗细不分、虚实不分，激光切割编程员还要手动删半天辅助线。

def _export_full_engineering_dxf(self, solid: cq.Workplane, manifest: dict, out_path: Path):
"""利用 OpenCascade HLR 投影生成多视图，并用 ezdxf 组装国标图层、尺寸标注与标题栏"""
doc = ezdxf.new("R2010", setup=True)
doc.units = units.MM
msp = doc.modelspace()
# 创建国标机械制图标准图层
doc.layers.add("CUT_PROFILE", color=7, lineweight=50)
doc.layers.add("HIDDEN_LINE", color=8, linetype="DASHED", lineweight=25)
doc.layers.add("DIMENSIONS", color=3, lineweight=18)
doc.layers.add("TITLE_BLOCK", color=5, lineweight=35)
bbox = manifest.get("cad_verification", {}).get("verified_bbox_mm") or manifest["target_bbox_mm"]
lx, ly, lz = float(bbox["x"]), float(bbox["y"]), float(bbox["z"])
# 定义 A3 图幅 (420mm x 297mm) 下四个视图的排布基准坐标
views = [
("FRONT (主视图)", (0, -1, 0), (100.0, 180.0)),
("TOP (俯视图)", (0, 0, 1), (100.0, 80.0)),
("LEFT (左视图)", (-1, 0, 0), (260.0, 180.0)),
("ISO (轴测参考)", (1, -1, 1), (280.0, 80.0)),
]
for view_name, direction, offset_xy in views:
self._project_shape_to_msp(solid, msp, direction, offset_xy)
msp.add_text(
view_name,
dxfattribs={"layer": "TITLE_BLOCK", "height": 3.5}
).set_placement((offset_xy[0] - lx * 0.3, offset_xy[1] - max(ly, lz) * 0.75))
# 在主视图与俯视图上自动打上三轴关键检验总尺寸 (总长 X、总高 Z、总宽 Y)
fx, fy = 100.0, 180.0
# 主视图底部标注总长 X
dim_x = msp.add_linear_dim(
base=(fx, fy - lz / 2 - 12),
p1=(fx - lx / 2, fy - lz / 2),
p2=(fx + lx / 2, fy - lz / 2),
dimstyle="EZDXF",
override={"dimtxt": 3.0, "dimasz": 2.5},
dxfattribs={"layer": "DIMENSIONS"}
)
dim_x.render()
# 主视图左侧标注总高 Z
dim_z = msp.add_linear_dim(
base=(fx - lx / 2 - 12, fy),
p1=(fx - lx / 2, fy - lz / 2),
p2=(fx - lx / 2, fy + lz / 2),
angle=90,
dimstyle="EZDXF",
override={"dimtxt": 3.0, "dimasz": 2.5},
dxfattribs={"layer": "DIMENSIONS"}
)
dim_z.render()
# 俯视图左侧标注总宽 Y
tx, ty = 100.0, 80.0
dim_y = msp.add_linear_dim(
base=(tx - lx / 2 - 12, ty),
p1=(tx - lx / 2, ty - ly / 2),
p2=(tx - lx / 2, ty + ly / 2),
angle=90,
dimstyle="EZDXF",
override={"dimtxt": 3.0, "dimasz": 2.5},
dxfattribs={"layer": "DIMENSIONS"}
)
dim_y.render()
# 绘制 A3 标准外图框 (420x297) 与右下角工程标题栏 + 技术要求
self._draw_a3_title_block(msp, manifest, lx, ly, lz)
doc.saveas(str(out_path))
print(f"  ├─ [工程总图] 带图层与尺寸标注的 A3 DXF 已导出: {out_path.name}")
def _project_shape_to_msp(self, solid: cq.Workplane, msp, proj_dir: tuple, offset: tuple):
"""调用 OpenCascade HLR 算法将 3D 实体投射为 2D 可见实线与隐藏虚线，并平移至指定图幅位置"""
hlr_Result = cq.occ_impl.exporters.svg.getProjection(solid.val(), proj_dir, (0, 0, 1))
ox, oy = offset
# 1. 绘制可见实线轮廓 -> CUT_PROFILE 层
for edge in hlr_Result["visible"].Edges():
pts = self._discretize_edge(edge, ox, oy)
if len(pts) >= 2:
msp.add_lwpolyline(pts, dxfattribs={"layer": "CUT_PROFILE"})
# 2. 绘制内部隐藏特征虚线 -> HIDDEN_LINE 层
for edge in hlr_Result["hidden"].Edges():
pts = self._discretize_edge(edge, ox, oy)
if len(pts) >= 2:
msp.add_lwpolyline(pts, dxfattribs={"layer": "HIDDEN_LINE"})
@staticmethod
def _discretize_edge(edge, ox: float, oy: float, num_pts: int = 24) -> list[tuple[float, float]]:
"""将 OCC 曲线离散化为高精度 2D 坐标点序列"""
try:
params = [edge.FirstParameter() + (edge.LastParameter() - edge.FirstParameter()) * i / (num_pts - 1) for i in range(num_pts)]
return [(round(edge.valueAt(u).x + ox, 3), round(edge.valueAt(u).y + oy, 3)) for u in params]
except Exception:
return []
@staticmethod
def _draw_a3_title_block(msp, manifest: dict, lx: float, ly: float, lz: float):
"""绘制 A3 图框 (420x297mm) 并自动注入来自 Phase 1 RAG 的车间制造技术要求"""
# 外框与内框
msp.add_lwpolyline([(0, 0), (420, 0), (420, 297), (0, 297), (0, 0)], dxfattribs={"layer": "TITLE_BLOCK"})
msp.add_lwpolyline([(10, 10), (410, 10), (410, 287), (10, 287), (10, 10)], dxfattribs={"layer": "TITLE_BLOCK"})
# 右下角标题栏区域
msp.add_lwpolyline([(250, 10), (410, 10), (410, 55), (250, 55), (250, 10)], dxfattribs={"layer": "TITLE_BLOCK"})
mfg = manifest.get("manufacturing_constraints", {})
vol = manifest.get("cad_verification", {}).get("volume_mm3", 0.0)
tech_notes = [
f"PROJECT ID : {manifest.get('project_id', 'HD-001')}",
f"MATERIAL   : {mfg.get('material', 'PC+ABS / AL6061')}",
f"PROCESS    : {mfg.get('process', 'Injection Molding / CNC')}",
f"WALL / DRAFT: {mfg.get('min_wall_thickness_mm', 2.4)}mm / >= {mfg.get('min_draft_angle_deg', 1.8)} deg",
f"BBOX / VOL : {lx}x{ly}x{lz} mm | {round(vol/1000.0, 2)} cm3",
"TOLERANCE  : GB/T 1804-m (Unmarked Fillet R0.5)",
]
for idx, line in enumerate(tech_notes):
msp.add_text(line, dxfattribs={"layer": "TITLE_BLOCK", "height": 2.8}).set_placement((255, 48 - idx * 6.2))
@staticmethod
def _render_dxf_to_pdf(dxf_path: Path, pdf_path: Path):
"""将生成的 DXF 渲染为车间可直接打印审阅的矢量 PDF"""
doc = ezdxf.readfile(str(dxf_path))
msp = doc.modelspace()
fig = plt.figure(figsize=(16.53, 11.69), dpi=300)  # 标准 A3 比例
ax = fig.add_axes([0, 0, 1, 1])
ctx = RenderContext(doc)
out = MatplotlibBackend(ax)
Frontend(ctx, out).draw_layout(msp, finalize=True)
fig.savefig(str(pdf_path), format="pdf")
plt.close(fig)
print(f"  └─ [移动审阅] 矢量工程图 PDF 已导出: {pdf_path.name}")

四、 如何对接你本地的浩辰 CAD / 中望 CAD / AutoCAD？（MCP 双轨策略）
你在规划里非常清醒地指出：“MCP 只是桥梁，不能白嫖 AutoCAD 本体”。在实际生产中，我建议你采用 “95% 靠上面的 ezdxf 无头直出 + 5% 靠国产 CAD COM 接口交互微调” 的方案：
为什么浩辰/中望/AutoCAD 可以共用一套本地控制逻辑？
无论是 AutoCAD、中望 CAD（ZWCAD）还是浩辰 CAD（GstarCAD），它们在 Windows 系统底层都完整实现了标准的 ActiveX / COM 自动化接口（通过 Python 的 pywin32 即可直接调起正在运行的 CAD 窗口）。
唯一的区别只是 ProgID 名字不同：
AutoCAD: "AutoCAD.Application"
中望 CAD: "ZWCAD.Application"
浩辰 CAD: "GCAD.Application"
极简通用国产 CAD 桥接代码（无需受制于特定第三方 MCP 版本）：
如果你电脑上装了浩辰或中望 CAD，想让大模型自动打开刚刚生成的 engineering_A3.dxf 并追加标注或导出打印，只需在 MCP Server 里放这 15 行自适应代码：

一、 双层自动评判引擎：客观几何铁律 > VLM 主观审美
在进入人工打勾（Phase 4 审阅台）之前，系统每天夜间可以无人值守跑出几十组候选方案。为了不让烂方案浪费你的精力，我们先用 phase7_judge_pipeline.py 进行两级漏斗过滤：
第一关（一票否决权）：客观几何与 DFM 规则硬校验
不相信任何大模型的“嘴上保证”，直接用 trimesh 射线测厚法（Ray-Casting）和 OpenCascade 拓扑算子对生成的 3D 模型做物理体检：
水密性与流形检查（Watertight & Manifold）：存在破洞或自交面的网格直接淘汰；
包络长宽比偏差（Proportion Drift）：白模/CAD 尺寸与需求包络偏差超过 8% 直接淘汰；
壁厚红线扫描（Min Wall Thickness）：从零件表面向内发射法向射线，测量真实局部壁厚。若二号车间要求 $\ge 2.4\text{mm}$，而实测存在 $< 1.8\text{mm}$ 的薄壁区（注塑欠注风险）或 $> 4.5\text{mm}$ 的肥厚积料区（缩水纹风险），直接判定为 rejected；
拔模角扫描（Draft Angle Check）：统计所有近似平行于脱模方向（Z 轴）的立面法向量夹角，识别零度直壁（卡模拉伤风险）。
第二关（造型与特征对齐）：本地 VLM 视觉初筛
只有通过第一关几何硬指标的方案，才会把渲染图送入 Ollama 里的 qwen2.5-vl:7b（或 glm-4v），检查视觉特征是否漏项（如：“需求要求的顶部状态指示灯导光窗口、两侧 M6 安装耳是否在图中清晰可见？”）。

# D:\hyperdesign\src\phase7_dfm_and_vlm_judge.pyimport jsonimport base64import numpy as npimport trimeshimport ollamafrom pathlib import Path
VLM_MODEL = "qwen2.5-vl:7b"
class HyperDesignJudge:
def __init__(self, project_dir: Path):
self.project_dir = project_dir
self.manifest_path = project_dir / "manifest.json"
def evaluate_geometry_dfm(self) -> dict:
"""第一优先级：客观几何与 DFM 规则硬校验（射线测厚 + 拔模角 + 包络误差）"""
with open(self.manifest_path, "r", encoding="utf-8") as f:
manifest = json.load(f)
# 将 Phase 5 产出的 STEP 离散化或读取高精度 STL 进行法向射线分析
stl_path = self.project_dir / manifest["mesh_metrics"]["clean_stl"]
mesh = trimesh.load_mesh(str(stl_path))
mfg = manifest.get("manufacturing_constraints", {})
min_wall_req = float(mfg.get("min_wall_thickness_mm", 2.0))
min_draft_req = float(mfg.get("min_draft_angle_deg", 1.5))
# 1. 随机采样 500 个表面点向内发射反向法线射线，计算真实局部壁厚分布
pts, face_idx = trimesh.sample.sample_surface(mesh, 500)
inward_normals = -mesh.face_normals[face_idx]
locations, index_ray, _ = mesh.ray.intersects_location(
ray_origins=pts + inward_normals * 0.05,
ray_directions=inward_normals,
multiple_hits=False
)
if len(locations) > 0:
hit_origins = pts[index_ray]
thicknesses = np.linalg.norm(locations - hit_origins, axis=1)
p10_thickness = float(np.percentile(thicknesses, 10))  # 10%分位最薄处壁厚
else:
p10_thickness = min_wall_req
# 2. 拔模角检查：查找接近垂直立面（法线 Z 分量接近 0）的实际倾角
z_dots = np.abs(mesh.face_normals[:, 2])
vertical_wall_mask = z_dots < np.sin(np.radians(10.0))  # 筛选倾角小于10度的立面
if np.any(vertical_wall_mask):
draft_angles_deg = np.degrees(np.arcsin(z_dots[vertical_wall_mask]))
median_draft = float(np.median(draft_angles_deg))
else:
median_draft = 90.0
# 3. 汇总硬性判定结果
wall_ok = p10_thickness >= (min_wall_req * 0.85)
draft_ok = median_draft >= (min_draft_req * 0.8)
dfm_passed = bool(mesh.is_watertight and wall_ok and draft_ok)
report = {
"is_watertight": bool(mesh.is_watertight),
"measured_p10_wall_mm": round(p10_thickness, 2),
"required_min_wall_mm": min_wall_req,
"measured_median_draft_deg": round(median_draft, 2),
"required_min_draft_deg": min_draft_req,
"dfm_hard_passed": dfm_passed
}
print(f"[DFM 硬规则检验] 通过={dfm_passed} | 实测P10壁厚={report['measured_p10_wall_mm']}mm | 立面拔模角={report['measured_median_draft_deg']}°")
return report
def evaluate_vlm_concepts(self, brief_text: str) -> list[dict]:
"""第二优先级：调用本地 VLM 对 Phase 2 概念图做工业结构合理性初筛打分"""
with open(self.manifest_path, "r", encoding="utf-8") as f:
manifest = json.load(f)

prompt = (
f"请作为工业设计总监审阅这张产品概念图。原始设计需求为：{brief_text}\n"

)
resp = ollama.chat(
model=VLM_MODEL,
format="json",
messages=[{"role": "user", "content": prompt, "images": [img_b64]}],

score_data = json.loads(resp["message"]["content"])
cand["vlm_score"] = score_data

# /mnt/d/hyperdesign/src/phase7_unsloth_dpo_trainer.pyimport osfrom datetime import datetimefrom pathlib import Pathimport torchfrom datasets import load_datasetfrom trl import DPOTrainer, DPOConfigfrom unsloth import FastLanguageModel, PatchDPOTrainer
# 激活 Unsloth 针对 DPO 的显存与算子加速补丁
PatchDPOTrainer()
# ================= 路径与版本配置 =================
BASE_MODEL = "unsloth/Qwen2.5-7B-Instruct-bnb-4bit"  # 或本地 Qwen3.5 / Gemma4 基座路径
DATASET_FILES = [
"/mnt/d/hyperdesign/models/checkpoints/dpo_dataset/phase2_prompt_dpo.jsonl",
"/mnt/d/hyperdesign/models/checkpoints/dpo_dataset/phase5_cad_code_dpo.jsonl"
]
VERSION_TAG = datetime.now().strftime("v%Y%m%d_%H%M")
OUTPUT_DIR = Path(f"/mnt/d/hyperdesign/models/checkpoints/lora_{VERSION_TAG}")
OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
def run_dpo_evolution():
print(f"[Phase 7 Evolution] 加载 4-bit 基座模型: {BASE_MODEL}")
model, tokenizer = FastLanguageModel.from_pretrained(
model_name=BASE_MODEL,
max_seq_length=4096,
dtype=None,
load_in_4bit=True,
)
# 挂载 LoRA 适配器（仅训练约 1~2% 的参数，零遗忘风险）
model = FastLanguageModel.get_peft_model(
model,
r=16,
target_modules=["q_proj", "k_proj", "v_proj", "o_proj", "gate_proj", "up_proj", "down_proj"],
lora_alpha=16,
lora_dropout=0,
bias="none",
use_gradient_checkpointing="unsloth",
random_state=3407,
)
# 加载平时工作流自动攒下的正负偏好样本对 (prompt, chosen, rejected)
valid_files = [f for f in DATASET_FILES if os.path.exists(f)]
if not valid_files:
raise FileNotFoundError("尚未检测到 DPO 数据集文件，请先运行 Phase 2~5 积累至少 50 条样本。")
dataset = load_dataset("json", data_files=valid_files, split="train")
print(f"[Phase 7 Evolution] 共加载 {len(dataset)} 组设计师优选/DFM校验 DPO 样本对。")
dpo_trainer = DPOTrainer(
model=model,
ref_model=None,  # Unsloth 会在内部零拷贝复用基座权重作为 Reference Model，省下一半显存
args=DPOConfig(
per_device_train_batch_size=2,
gradient_accumulation_steps=4,
warmup_ratio=0.1,
num_train_epochs=3,
learning_rate=5e-6,
fp16=not torch.cuda.is_bf16_supported(),
bf16=torch.cuda.is_bf16_supported(),
logging_steps=1,
optim="adamw_8bit",
weight_decay=0.0,
lr_scheduler_type="linear",
seed=42,
output_dir=str(OUTPUT_DIR),
beta=0.1,  # DPO 温度超参：控制偏离基座模型的幅度
max_length=2048,
max_prompt_length=1024,
),
train_dataset=dataset,
tokenizer=tokenizer,
)
print("[Phase 7 Evolution] 开始执行 QLoRA + DPO 偏好微调...")
dpo_trainer.train()
# 1. 保存轻量级 LoRA 适配器权重（仅几十 MB，随时可回滚）
model.save_pretrained(str(OUTPUT_DIR / "lora_adapter"))
tokenizer.save_pretrained(str(OUTPUT_DIR / "lora_adapter"))
# 2. 直接合并并量化导出为 Ollama 可用的 GGUF (Q4_K_M)，实现无缝生产迭代
gguf_dir = OUTPUT_DIR / "gguf"
print(f"[Phase 7 Evolution] 正在导出 GGUF 至 {gguf_dir}...")
model.save_pretrained_gguf(str(gguf_dir), tokenizer, quantization_method="q4_k_m")
print(f"[微调完成!] 演进版权重已存盘: {OUTPUT_DIR}")
print(f"请在 Windows 终端运行以下命令将其注册为新主控大脑：\n"
f"ollama create hd-brain:{VERSION_TAG} -f D:\\hyperdesign\\models\\checkpoints\\lora_{VERSION_TAG}\\gguf\\Modelfile")
if __name__ == "__main__":
run_dpo_evolution()
